% needspace, hand-rolled to keep the preamble dependency-free. Reserves #1 lines
% and moves the whole block to the next page if they are not there, so a role or
% project header never lands alone at the foot of a page.
\newcommand{\needlines}[1]{%
  \par\penalty-100%
  \begingroup\dimen0=#1\baselineskip
  \vskip\dimen0 \penalty9999 \vskip-\dimen0 \vskip 0pt%
  \endgroup%
}

% Headers are glued to whatever follows them: \nopagebreak[4] kills the break
% LaTeX would otherwise take at the list's \topsep glue.
% Jake's two-line subheading: bold organisation with location on the right,
% then the role (or degree) in italics with dates on the right.
\newcommand{\resumeSubheading}[4]{%
  \needlines{4}%
  \vspace{\entrygap}%
  \hspace*{0.15in}\textbf{#1} \hfill #2\\%
  \hspace*{0.15in}{\small\itshape #3} \hfill {\small\itshape #4}\par\nopagebreak[4]%
}

}
