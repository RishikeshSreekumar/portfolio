% needspace, hand-rolled to keep the preamble dependency-free. Reserves #1 lines
% and moves the whole block to the next page if they are not there, so a role or
% project header never lands alone at the foot of a page.
\newcommand{\needlines}[1]{%
  \par\penalty-100%
  \begingroup\dimen0=#1\baselineskip
  \vskip\dimen0 \penalty9999 \vskip-\dimen0 \vskip 0pt%
  \endgroup%
}

% A role or project header is glued to whatever follows it: \nopagebreak[4]
% kills the break LaTeX would otherwise take at the list's \topsep glue, which
% is what split "Workfence Technologies / Remote" from its own bullets.

  {\small #3} \hfill {\small #4}\par\nopagebreak[4]%
}
